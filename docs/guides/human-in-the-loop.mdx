Many agent scenarios require human oversight before executing high-risk operations. This guide shows how to implement human-in-the-loop patterns with Noēsis.

## When to use human approval

Add human approval when:

- Actions are **irreversible** (deletes, deployments, financial transactions)
- Risk scores **exceed thresholds**
- Policies flag operations as **requiring review**
- Regulations demand **audit trails** with human sign-off

## Basic approval pattern

Pause **during** the run, then continue the same episode ID. Do not call `ns.interrupt(...)` after a completed `ns.solve(...)` / `ns.run(...)`: terminal runs write `final.json` and reject lifecycle mutations with `RunSealedError`. Starting a follow-up `ns.run(...)` after approval is a **new** episode, not a continuation.

Canonical path: enforce-mode pause-on-veto.

```python
import noesis as ns
from pathlib import Path
from noesis.io import paths


def latest_checkpoint_id(episode_id: str) -> str | None:
    for event in ns.events.read(episode_id):
        if event.get("phase") == "runtime" and event.get("event_type") == "run.checkpoint":
            checkpoint_id = (event.get("payload") or {}).get("checkpoint_id")
            if isinstance(checkpoint_id, str) and checkpoint_id:
                return checkpoint_id
    return None


def is_sealed(episode_id: str) -> bool:
    return (Path(paths(episode_id)["dir"]) / "final.json").exists()


def require_approval(episode_id: str, action: str) -> bool:
    """Replace with Slack, ServiceNow, or your approval UI."""
    print(f"\nApproval required for episode {episode_id}")
    print(f"   Action: {action}")
    return input("   Approve? (y/n): ").lower() == "y"


def run_with_approval(task: str, using):
    ns.set(
        governance_mode="enforce",
        governance_pause_on_veto=True,
    )
    episode_id = ns.solve(task, using=using, intuition=True)

    if is_sealed(episode_id):
        return episode_id  # no veto / no pause

    checkpoint_id = latest_checkpoint_id(episode_id)
    if checkpoint_id is None:
        return episode_id

    if not require_approval(episode_id, task):
        print("Action rejected by human reviewer")
        return None

    return ns.resume_run(
        episode_id,
        checkpoint_id=checkpoint_id,
        using=using,
    )
```

<Tip>
Prefer `resume_run(...)` over rerunning from scratch after approval. It preserves a continuous, append-only run history. `resume()` emits `run.resume` evidence only and does not continue execution.
</Tip>

<Warning>
Paused runs stay unsealed: `final.json` and `manifest.json` are absent until continuation finishes. Do not hand-edit `events.jsonl`, `state.json`, or checkpoint files while waiting on approval (`CheckpointConsistencyError` on resume).
</Warning>

## Policy-driven approval

Create a policy that flags operations for approval:

```python
import noesis as ns
from noesis.intuition import IntuitionEvent


class ApprovalPolicy(ns.DirectedIntuition):
    """Policy that requires approval for high-risk operations."""
    
    __version__ = "1.0"
    
    HIGH_RISK_KEYWORDS = {"delete", "drop", "truncate", "deploy", "rollback"}
    
    def advise(self, state: dict) -> IntuitionEvent | None:
        task = state.get("task", "").lower()
        
        # Check for high-risk keywords
        if any(kw in task for kw in self.HIGH_RISK_KEYWORDS):
            return self.intervene(
                advice="High-risk operation detected. Requiring human approval.",
                patch={
                    "requires_approval": True,
                    "risk_reason": "high_risk_keyword",
                },
                target="plan",
                rationale="Operations matching high-risk patterns need review.",
            )
        
        # Check for production environment
        if "production" in task or state.get("environment") == "production":
            return self.intervene(
                advice="Production operation detected. Requiring human approval.",
                patch={
                    "requires_approval": True,
                    "risk_reason": "production_environment",
                },
                target="plan",
                rationale="All production changes require human sign-off.",
            )
        
        return None
```

## Slack integration

Send approval requests to Slack:

```python
import os
from slack_sdk import WebClient
from slack_sdk.errors import SlackApiError
import noesis as ns


class SlackApproval:
    """Handle approvals via Slack interactive messages."""
    
    def __init__(self, channel: str):
        self.client = WebClient(token=os.environ["SLACK_BOT_TOKEN"])
        self.channel = channel
        self.pending = {}  # episode_id -> message_ts
    
    def request_approval(self, episode_id: str, action: str, context: dict) -> str:
        """Send approval request to Slack."""
        blocks = [
            {
                "type": "header",
                "text": {"type": "plain_text", "text": "🔒 Approval Required"},
            },
            {
                "type": "section",
                "text": {
                    "type": "mrkdwn",
                    "text": f"*Episode:* `{episode_id}`\n*Action:* {action}",
                },
            },
            {
                "type": "context",
                "elements": [
                    {"type": "mrkdwn", "text": f"*Reason:* {context.get('reason', 'Policy requirement')}"},
                ],
            },
            {
                "type": "actions",
                "elements": [
                    {
                        "type": "button",
                        "text": {"type": "plain_text", "text": "✅ Approve"},
                        "style": "primary",
                        "action_id": f"approve_{episode_id}",
                    },
                    {
                        "type": "button",
                        "text": {"type": "plain_text", "text": "❌ Reject"},
                        "style": "danger",
                        "action_id": f"reject_{episode_id}",
                    },
                ],
            },
        ]
        
        try:
            response = self.client.chat_postMessage(
                channel=self.channel,
                blocks=blocks,
                text=f"Approval required for {episode_id}",
            )
            self.pending[episode_id] = response["ts"]
            return response["ts"]
        except SlackApiError as e:
            raise RuntimeError(f"Slack API error: {e.response['error']}")
    
    def handle_response(self, episode_id: str, approved: bool, user: str):
        """Handle approval response from Slack."""
        if episode_id in self.pending:
            # Update the message
            status = "✅ Approved" if approved else "❌ Rejected"
            self.client.chat_update(
                channel=self.channel,
                ts=self.pending[episode_id],
                text=f"{status} by {user}",
            )
            del self.pending[episode_id]
        
        return approved


# Usage — reuse latest_checkpoint_id from the basic pattern above
slack = SlackApproval(channel="#approvals")


def run_with_slack_approval(task: str, using):
    ns.set(governance_mode="enforce", governance_pause_on_veto=True)
    episode_id = ns.solve(task, using=using, intuition=True)

    checkpoint_id = latest_checkpoint_id(episode_id)
    if checkpoint_id is None:
        return {"status": "completed", "episode_id": episode_id}

    slack.request_approval(
        episode_id,
        task,
        {"reason": "governance veto", "checkpoint_id": checkpoint_id},
    )
    return {
        "status": "pending_approval",
        "episode_id": episode_id,
        "checkpoint_id": checkpoint_id,
    }


def on_slack_approved(episode_id: str, checkpoint_id: str, using):
    return ns.resume_run(episode_id, checkpoint_id=checkpoint_id, using=using)
```

## Async approval workflow

For production systems, use an async workflow:

```python
import asyncio
from dataclasses import dataclass
from enum import Enum
import noesis as ns


class ApprovalStatus(Enum):
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"
    EXPIRED = "expired"


@dataclass
class ApprovalRequest:
    episode_id: str
    action: str
    context: dict
    status: ApprovalStatus = ApprovalStatus.PENDING
    approver: str | None = None


class ApprovalQueue:
    """Async approval queue with timeout support."""
    
    def __init__(self, timeout_seconds: int = 3600):
        self.requests: dict[str, ApprovalRequest] = {}
        self.timeout = timeout_seconds
        self._events: dict[str, asyncio.Event] = {}
    
    async def request_approval(self, episode_id: str, action: str, context: dict) -> ApprovalRequest:
        """Submit an approval request and wait for response."""
        request = ApprovalRequest(episode_id, action, context)
        self.requests[episode_id] = request
        self._events[episode_id] = asyncio.Event()
        
        # Send to your approval channel (Slack, email, etc.)
        await self._notify_approvers(request)
        
        # Wait for response with timeout
        try:
            await asyncio.wait_for(
                self._events[episode_id].wait(),
                timeout=self.timeout,
            )
        except asyncio.TimeoutError:
            request.status = ApprovalStatus.EXPIRED
        
        return request
    
    async def approve(self, episode_id: str, approver: str):
        """Mark request as approved."""
        if episode_id in self.requests:
            self.requests[episode_id].status = ApprovalStatus.APPROVED
            self.requests[episode_id].approver = approver
            self._events[episode_id].set()
    
    async def reject(self, episode_id: str, approver: str):
        """Mark request as rejected."""
        if episode_id in self.requests:
            self.requests[episode_id].status = ApprovalStatus.REJECTED
            self.requests[episode_id].approver = approver
            self._events[episode_id].set()
    
    async def _notify_approvers(self, request: ApprovalRequest):
        """Send notification to approvers."""
        # Implement your notification logic here
        print(f"Approval requested: {request.episode_id}")


# Usage
queue = ApprovalQueue(timeout_seconds=300)


async def run_async_approval(task: str, using):
    ns.set(governance_mode="enforce", governance_pause_on_veto=True)
    episode_id = ns.solve(task, using=using, intuition=True)
    checkpoint_id = latest_checkpoint_id(episode_id)
    if checkpoint_id is None:
        return episode_id

    request = await queue.request_approval(episode_id, task, {"checkpoint_id": checkpoint_id})

    if request.status == ApprovalStatus.APPROVED:
        return ns.resume_run(
            episode_id,
            checkpoint_id=checkpoint_id,
            using=using,
        )
    if request.status == ApprovalStatus.EXPIRED:
        print("Approval timed out")
        return None
    print(f"Rejected by {request.approver}")
    return None
```

## Recording approval in events

Keep the decision on the **same** run. `resume_run` appends `run.resume` and continues execution. A follow-up `ns.run(...)` is a different episode and breaks the causal chain.

If you need an explicit audit note before continuation, emit it through your approval adapter or inspect the existing `run.interrupt` / `run.checkpoint` / `run.resume` events rather than opening a new episode.

## UI for approvals

Build a simple approval UI with Gradio. List **unsealed** runs that already have a checkpoint, then call `resume_run` (do not start a new episode):

```python
import gradio as gr
import noesis as ns
from pathlib import Path
from noesis.io import list_runs, paths


def get_pending_approvals():
    pending = []
    for ep in list_runs(limit=50):
        episode_id = ep["episode_id"]
        if (Path(paths(episode_id)["dir"]) / "final.json").exists():
            continue
        checkpoint_id = latest_checkpoint_id(episode_id)
        if checkpoint_id is None:
            continue
        pending.append({
            "episode_id": episode_id,
            "task": ep.get("task", ""),
            "started_at": ep.get("started_at"),
            "checkpoint_id": checkpoint_id,
        })
    return pending


def approve_episode(episode_id: str, using):
    checkpoint_id = latest_checkpoint_id(episode_id)
    if checkpoint_id is None:
        return "No checkpoint on this episode (already sealed or not paused)"
    ns.resume_run(episode_id, checkpoint_id=checkpoint_id, using=using)
    return f"Resumed {episode_id}"


def reject_episode(episode_id: str, reason: str):
    return (
        f"Left {episode_id} paused. Rejection reason: {reason}. "
        "Do not call resume_run; the run stays unsealed until you continue or abandon it."
    )


with gr.Blocks(title="Approval Queue") as app:
    gr.Markdown("# Approval Queue")
    using_input = gr.Textbox(label="Adapter import path (required for non-minimal runs)")
    refresh_btn = gr.Button("Refresh")
    pending_list = gr.Dataframe(headers=["Episode ID", "Task", "Started at", "Checkpoint"])
    episode_input = gr.Textbox(label="Episode ID")
    reason_input = gr.Textbox(label="Rejection reason (if rejecting)")
    approve_btn = gr.Button("Approve", variant="primary")
    reject_btn = gr.Button("Reject", variant="stop")
    result = gr.Textbox(label="Result")

    refresh_btn.click(
        lambda: [
            [p["episode_id"], p["task"], p["started_at"], p["checkpoint_id"]]
            for p in get_pending_approvals()
        ],
        outputs=pending_list,
    )
    approve_btn.click(
        lambda episode_id, using: approve_episode(episode_id, using),
        inputs=[episode_input, using_input],
        outputs=result,
    )
    reject_btn.click(reject_episode, inputs=[episode_input, reason_input], outputs=result)


if __name__ == "__main__":
    app.launch()
```

## Best practices

<Tip>
**Set reasonable timeouts.** Pending approvals shouldn't block indefinitely—expire them after a reasonable period.
</Tip>

<Warning>
**Authenticate approvers.** In production, verify the identity of approvers through SSO or similar.
</Warning>

<Info>
**Create audit trails.** Record all approval decisions with timestamps, approvers, and reasons in the event timeline.
</Info>
